cmake_minimum_required(VERSION 3.16)
project(emo_open_fw_host_tests C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(LINK_DIR ${CMAKE_CURRENT_LIST_DIR}/../../components/k210_link)
file(GLOB LINK_SRCS ${LINK_DIR}/*.c)

set(MOTION_DIR ${CMAKE_CURRENT_LIST_DIR}/../../components/motion)
file(GLOB MOTION_SRCS ${MOTION_DIR}/*.c)
add_library(motion_host STATIC ${MOTION_SRCS})
target_include_directories(motion_host PUBLIC ${MOTION_DIR}/include)
target_compile_options(motion_host PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_libraries(motion_host PUBLIC m)
target_link_options(motion_host PUBLIC -fsanitize=address,undefined)

set(MOT_DIR ${CMAKE_CURRENT_LIST_DIR}/../../components/mot)
file(GLOB MOT_SRCS ${MOT_DIR}/*.c)
add_library(mot_host STATIC ${MOT_SRCS})
target_include_directories(mot_host PUBLIC ${MOT_DIR}/include)
target_compile_options(mot_host PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_libraries(mot_host PUBLIC m)
target_link_options(mot_host PUBLIC -fsanitize=address,undefined)

add_library(k210_link_host STATIC ${LINK_SRCS})
target_include_directories(k210_link_host PUBLIC ${LINK_DIR}/include PRIVATE ${LINK_DIR})
target_compile_options(k210_link_host PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(k210_link_host PUBLIC -fsanitize=address,undefined)

enable_testing()

foreach(t test_crc test_frames test_rx test_link test_media test_spi test_loopback)
    add_executable(${t} ${t}.c)
    target_link_libraries(${t} PRIVATE k210_link_host)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_LIST_DIR})
    target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
    add_test(NAME ${t} COMMAND ${t})
endforeach()

set(SEAM_DIR ${CMAKE_CURRENT_LIST_DIR}/../../components/seam)
file(GLOB SEAM_SRCS ${SEAM_DIR}/*.c)
add_library(seam_host STATIC ${SEAM_SRCS})
target_include_directories(seam_host PUBLIC ${SEAM_DIR}/include)
target_compile_options(seam_host PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_libraries(seam_host PUBLIC k210_link_host)

add_executable(test_seam test_seam.c)
target_link_libraries(test_seam PRIVATE seam_host)
target_include_directories(test_seam PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(test_seam PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
add_test(NAME test_seam COMMAND test_seam)

add_executable(test_mot test_mot.c)
target_link_libraries(test_mot PRIVATE mot_host)
target_include_directories(test_mot PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(test_mot PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
add_test(NAME test_mot COMMAND test_mot)

add_executable(test_osc test_osc.c)
target_link_libraries(test_osc PRIVATE motion_host)
target_include_directories(test_osc PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(test_osc PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
add_test(NAME test_osc COMMAND test_osc)

# Repository-level checks
find_program(GO go)

    set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
    add_test(NAME server_go COMMAND ${GO} test -count=1 ./... WORKING_DIRECTORY ${ROOT}/server)
    add_test(NAME tools_go COMMAND ${GO} test -count=1 ./... WORKING_DIRECTORY ${ROOT}/tools/emotool)
    add_test(NAME denylist_repo COMMAND ${GO} run . denylist-scan WORKING_DIRECTORY ${ROOT}/tools/emotool)
    add_test(NAME partitions COMMAND ${GO} run . partitions ${ROOT}/partitions.csv WORKING_DIRECTORY ${ROOT}/tools/emotool)
